cmake_minimum_required(VERSION 3.20)
project(Lab2 LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "Windows is required")
endif()
add_executable(Lab2 WIN32 main.cpp Shape.cpp PointShape.cpp LineShape.cpp RectShape.cpp EllipseShape.cpp)
target_compile_features(Lab2 PRIVATE cxx_std_17)
target_compile_definitions(Lab2 PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(Lab2 PRIVATE user32 gdi32)
if(MSVC)
    target_compile_options(Lab2 PRIVATE /W4 /utf-8)
else()
    target_compile_options(Lab2 PRIVATE -Wall -Wextra -Wpedantic)
    target_link_options(Lab2 PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()

option(LAB2_BUILD_TESTS "Build integration checks" OFF)
if(LAB2_BUILD_TESTS)
    enable_testing()
    add_executable(Lab2Smoke tests/smoke.cpp Shape.cpp PointShape.cpp LineShape.cpp RectShape.cpp EllipseShape.cpp)
    target_compile_features(Lab2Smoke PRIVATE cxx_std_17)
    target_compile_definitions(Lab2Smoke PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(Lab2Smoke PRIVATE user32 gdi32)
    if(MSVC)
        target_compile_options(Lab2Smoke PRIVATE /W4 /utf-8 /UNDEBUG)
    else()
        target_compile_options(Lab2Smoke PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
        target_link_options(Lab2Smoke PRIVATE -static)
    endif()
    add_test(NAME Lab2Smoke COMMAND Lab2Smoke)
endif()
